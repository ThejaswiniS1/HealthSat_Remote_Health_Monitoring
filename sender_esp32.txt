#include <SPI.h>
#include <LoRa.h>
#include <Wire.h>
#include "MAX30105.h"

// --- LoRa Pin Definitions ---
#define LORA_SCK 18
#define LORA_MISO 19
#define LORA_MOSI 23
#define LORA_CS 5
#define LORA_RST 14
#define LORA_DIO0 2

// --- Sensor Object & Variables ---
MAX30105 particleSensor;
long lastBeat = 0;
float heartRate;
int beatThreshold = 8000;
bool beatDetected = false;

void setup() {
  Serial.begin(9600);
  Serial.println("LoRa Sender with MAX30105");

  // 1. SENSOR SETUP
  if (!particleSensor.begin(Wire, I2C_SPEED_FAST)) {
    Serial.println("MAX30105 not found.");
    while (1);
  }
  particleSensor.setup();
  Serial.println("Sensor Ready.");

  // 2. LORA SETUP
  LoRa.setPins(LORA_CS, LORA_RST, LORA_DIO0);
  // Set frequency to 866MHz for India/Europe. Check your module's frequency!
  if (!LoRa.begin(866E6)) { 
    Serial.println("Starting LoRa failed!");
    while (1);
  }
  Serial.println("LoRa Ready. Sending data...");
}

void loop() {
  long irValue = particleSensor.getIR();

  // Calculate BPM (same logic as before)
  if (irValue > beatThreshold) {
    if (!beatDetected) {
      long timeDifference = millis() - lastBeat;
      heartRate = 60.0 * 1000.0 / timeDifference;
      lastBeat = millis();
      beatDetected = true;
    }
  } else {
    beatDetected = false;
  }
  
  // Format the data into a simple string: "BPM,IR_VALUE"
  String dataToSend = "";
  if (heartRate > 40 && heartRate < 200) {
      dataToSend += String(heartRate);
  } else {
      dataToSend += String(0);
  }
  dataToSend += ",";
  dataToSend += String(irValue);

  // Send the data as a LoRa packet
  LoRa.beginPacket();
  LoRa.print(dataToSend);
  LoRa.endPacket();

  Serial.println("Sending packet: " + dataToSend);

  delay(500); // Send data every half second
}
